GPS Arrival Detection
javascript// Pseudo-code for arrival detection
const checkArrival = (userLat, userLng, customerLat, customerLng) => {
  const distance = calculateDistance(userLat, userLng, customerLat, customerLng);
  if (distance <= 50) { // 50 meters
    showArrivalModal();
  }
};

// Use Haversine formula for distance calculation
const calculateDistance = (lat1, lon1, lat2, lon2) => {
  const R = 6371e3; // Earth radius in meters
  const φ1 = lat1 * Math.PI/180;
  const φ2 = lat2 * Math.PI/180;
  const Δφ = (lat2-lat1) * Math.PI/180;
  const Δλ = (lon2-lon1) * Math.PI/180;

  const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
          Math.cos(φ1) * Math.cos(φ2) *
          Math.sin(Δλ/2) * Math.sin(Δλ/2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

  return R * c; // Distance in meters
};

Progress Tracking
javascript// Calculate daily progress
const getDailyProgress = async (userId, date) => {
  const route = await getRouteByUserId(userId);
  const totalCustomers = route.customer_order.length;
  
  const visits = await getVisitsByUserAndDate(userId, date);
  const completedVisits = visits.filter(v => v.completed).length;
  
  return {
    total: totalCustomers,
    completed: completedVisits,
    remaining: totalCustomers - completedVisits,
    percentage: (completedVisits / totalCustomers) * 100
  };
